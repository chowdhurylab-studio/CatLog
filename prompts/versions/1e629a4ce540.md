=== .claude/workflows/catlog-batch.js ===
export const meta = {
  name: 'catlog-batch',
  description: 'CatLog v2: read and independently verify a batch of papers (args.lanes papers at a time, default 6)',
  whenToUse: 'After `python catlog.py next N` prints PMIDs; pass args.pmids (and args.lanes = the number the user gave, default 6; args.dbcheck = the PMIDs `python3 catlog.py dbdiff --pending N` prints and claims). Then run `python catlog.py add <PMIDs>` and `python3 catlog.py dbcheck-record`.',
  phases: [{ title: 'Papers', detail: 'read -> verify -> verify missed, one lane per paper in flight' },
           { title: 'Database differences', detail: 'args.dbcheck: one agent per task file (catlog-dbcheck), after the readers' }],
}

const DIR = '<the CatLog folder>'
const pmids = (Array.isArray(args) ? args : (args && args.pmids) || []).map(String)
if (!pmids.length) return { error: 'pass args.pmids (from: python catlog.py next N)' }
// args.dbcheck: papers whose values differ from a database and wait for a check against the paper (`catlog.py dbdiff
// --pending N`: task files an earlier batch's dbdiff wrote, claimed for this batch); run by the saved workflow
// catlog-dbcheck after the readers
const dbcheck = ((args && args.dbcheck) || []).map(String)
if (dbcheck.some(p => !/^(\d{1,9}|NP[0-9A-F]{8})$/.test(p))) return { error: 'args.dbcheck must be paper ids (from: python3 catlog.py dbdiff --pending N)' }

// args.lean: readers and verifiers run as the lean `catlog-reader` agent type (.claude/agents: only the file and shell
// tools, so each agent starts from far less context and the usage limit lasts longer). Off by default until tested;
// a session that does not know the type falls back to the normal agent.
const LEAN = !!(args && args.lean)
const run = async (prompt, opts) => {
  if (!LEAN) return agent(prompt, opts)
  try {
    return await agent(prompt, { ...opts, agentType: 'catlog-reader' })
  } catch (e) {
    if (/not found/.test(String(e))) return agent(prompt, opts)
    throw e
  }
}

const DONE = {
  type: 'object',
  properties: {
    pmid: { type: 'string' },
    rows: { type: 'integer', description: 'rows written (read) or verdicts written (verify)' },
    missed_count: { type: 'integer', description: 'verify only: rows written to the missed file, else 0' },
    problem: { type: 'string', description: 'empty if fine; else why the file could not be written' },
  },
  required: ['pmid', 'rows', 'missed_count', 'problem'],
}

// Leading with what the work is keeps the model's safety filter from stopping readers of toxin/pathogen enzyme papers.
const CONTEXT = 'Scientific data curation for a enzyme-kinetics research catalog: transcribe published kcat/Km values from one paper. '
const SLICES = 'If the paper is long, read it in slices of a few pages. '
// Parallel agents of one chat share its scratchpad: on 2026-10-09 the reader of 1857377 ran build.py after the reader of
// 17692452 had saved its own build.py under that name, and so rewrote 17692452's read file. Each agent keeps its
// scratch files in a folder named by its role and paper.
const SCRATCH = (role, pmid) => `Keep any scratch file you make (scripts, notes, rendered page images) in a folder of ` +
  `your own named ${role}-${pmid} inside your scratchpad directory (or /tmp), never under a shared name such as ` +
  `build.py: agents for other papers work in the same directory at the same time. `

const read = (pmid, model) => run(
  CONTEXT + SLICES + `Follow ${DIR}/prompts/read.md for ${/^\d+$/.test(pmid) ? 'PMID' : 'paper (no PMID: identified by its DOI in papers.tsv column 2)'} ${pmid}. The paper text is ${DIR}/texts/${pmid}.txt (read all of it, in chunks if long), ` +
  `plus ${DIR}/texts/${pmid}.supp.txt if that file exists (supplementary material). ` +
  `Write ${DIR}/work/${pmid}.read.json. Validate it with: python3 -c "import json;json.load(open('${DIR}/work/${pmid}.read.json'))". ` +
  `If that file already exists, an interrupted batch kept this paper's finished read (catlog.py reread keeps a read ` +
  `nobody verified yet): do not read the paper again - validate the file as above, leave it exactly as it is, and ` +
  `report its row count, so only the verifier runs. ` + SCRATCH('read', pmid) +
  `Do not edit any other file. Report the row count.`,
  { label: `read${model ? '-' + model : ''}:${pmid}`, phase: 'Papers', schema: DONE, ...(model ? { model } : {}) })

const verify = (pmid, which, model) => run(
  CONTEXT + SLICES + `Follow ${DIR}/prompts/verify.md for ${/^\d+$/.test(pmid) ? 'PMID' : 'paper (no PMID: identified by its DOI in papers.tsv column 2)'} ${pmid}. Paper: ${DIR}/texts/${pmid}.txt (and ${DIR}/texts/${pmid}.supp.txt if it exists). ` +
  `Rows to check: ${DIR}/work/${pmid}${which}.read.json. Write your verdicts to ${DIR}/work/${pmid}${which}.verify.json` +
  (which ? ' and do NOT write any missed file (set missed_count 0).' : '.') +
  ` You did not write these rows; check each against the paper yourself. Validate the JSON with python3 -c "import json;json.load(open(...))". ` +
  SCRATCH(which ? 'verify-missed' : 'verify', pmid) + `Do not edit any other file${which ? '' : ' (the missed file above is allowed)'}.`,
  { label: `verify${which || ''}${model ? '-' + model : ''}:${pmid}`, phase: 'Papers', schema: DONE, ...(model ? { model } : {}) })

// An agent that died (safety-filter stop, usage limit) returns null, or reports a `problem` when the stop hit one of its
// tool calls (19261311, 2026-10-09); retrying the same model on the same paper fails the same way, so the one retry runs
// on another model.
// Three such steps in a row across all lanes means the session's usage limit is gone, not a bad paper (24 papers were
// re-read for nothing on 2026-10-10): the lanes then stop taking new papers. A valid result with zero rows is fine.
let dead = 0
const DEAD_STOP = 3
const stopped = () => dead >= DEAD_STOP
const again = async (step) => {
  const a = await step()
  const r = a && !a.problem ? a : ((await step('sonnet')) || a)
  dead = r && !r.problem ? 0 : dead + 1
  return r
}

async function one(pmid) {
  const r = await again((model) => read(pmid, model))
  if (!r || r.problem) return { pmid, read: r, note: r ? 'read problem' : 'reader died twice: paper left as reading' }
  const v = await again((model) => verify(pmid, '', model))  // also for zero rows: confirms the paper prints no values
  let m = null
  if (v && v.missed_count > 0) m = await again((model) => verify(pmid, '.missed', model))
  return { pmid, read: r.rows, verify: v, missed: m }
}

// lanes, each working through its share in order: never more than `lanes` agents live (the user's number, default 6)
const LANES = Math.max(1, parseInt(args && args.lanes, 10) || 6)
const lanes = Array.from({ length: Math.min(LANES, pmids.length) }, (_, k) => async () => {
  const out = []
  for (let i = k; i < pmids.length; i += LANES) {
    if (stopped()) break
    out.push(await one(pmids[i]))
    log(`${pmids[i]} done`)
  }
  return out
})
const results = (await parallel(lanes)).filter(Boolean).flat()
const done = new Set(results.map(r => r.pmid))
// database differences: one dbcheck agent per task file (same lanes, same Sonnet retry and usage-limit stop), only
// after this batch's readers and verifiers and never once the usage limit stopped them
let checks = null
if (dbcheck.length && !stopped()) {
  phase('Database differences')
  try {
    checks = await workflow('catlog-dbcheck', { pmids: dbcheck, lanes: LANES })
  } catch (e) {
    checks = { error: `catlog-dbcheck did not run (${String(e).slice(0, 200)}): run it on its own with args.pmids = ${JSON.stringify(dbcheck)}` }
  }
}
return { results, add: `python3 ${DIR}/catlog.py add ${pmids.join(' ')}`,
  ...(dbcheck.length ? { dbcheck: checks || { not_started: dbcheck }, dbcheck_record: `python3 ${DIR}/catlog.py dbcheck-record` } : {}),
  ...(stopped() ? { stop: `${DEAD_STOP} agents in a row returned nothing (session usage limit): the lanes stopped taking new papers`,
                    not_started: pmids.filter(p => !done.has(p)) } : {}) }

---
name: catlog-reader
description: CatLog v2 paper reader or verifier (used by the catlog-batch workflow). Reads one paper from <the CatLog folder>/texts and writes the one JSON file its task names.
tools: Read, Write, Edit, Bash, Grep, Glob
model: inherit
---
You transcribe or check published enzyme kinetic values (kcat, Km, kcat/Km) for CatLog, a scientific research catalog
in <the CatLog folder>. Follow the prompt file your task names (prompts/read.md or prompts/verify.md)
exactly; it holds the scientific rules.

- Use only the paper's own files: texts/<id>.txt, texts/<id>.supp.txt, the original PDF or XML named in column 5 of
  papers.tsv (`cut -f1-5 papers.tsv | grep "^<id>[[:space:]]"`), and page images you render with pdftoppm into /tmp.
- Never open database values (work/dbcheck_*, work/old/dbcheck_* included), catlog_wide.*, earlier-catalog files or
  human_review/ (CatLog reads blind).
- Long files: read them in slices with Read (offset/limit); find tables with grep -n.
- Write only the output file your task names, validate it as the task says, and return the structured result it asks for.
